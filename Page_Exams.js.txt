        function loadExamsFromSheet(isForceRefresh = false) {
            if (isForceRefresh) {

        // 動態連動「年級」與「學校」選單
        function updateGradeAndSchoolOptions() {
            const currentLevel = levelSelect ? levelSelect.value : '';
            const currentSelectedCity = citySelect ? citySelect.value : '';
            const currentSelectedSchool = schoolSelect ? schoolSelect.value : '';
            const currentSelectedGrade = gradeSelect ? gradeSelect.value : '';

            gradeSelect.innerHTML = '<option value="">全部年級</option>';
            let grades = [];
            if (currentLevel === '國小') grades = ['小一', '小二', '小三', '小四', '小五', '小六'];
            else if (currentLevel === '國中') grades = ['國一', '國二', '國三'];
            else if (currentLevel === '高中') grades = ['高一', '高二', '高三'];
            else grades = ['小一', '小二', '小三', '小四', '小五', '小六', '國一', '國二', '國三', '高一', '高二', '高三'];

            grades.forEach(g => {
                const opt = document.createElement('option');
                opt.value = g;
                opt.textContent = g;
                if (g === currentSelectedGrade) opt.selected = true;
                gradeSelect.appendChild(opt);
            });

            schoolSelect.innerHTML = '<option value="">全部學校</option>';
            const filteredExams = exams.filter(e => {
                if (currentLevel && e.educationLevel !== currentLevel) return false;
                if (currentSelectedCity && e.city !== currentSelectedCity) return false;
                return true;
            });

            const schools = [...new Set(filteredExams.map(e => e.school))].filter(Boolean);
            schools.forEach(sch => {
                const opt = document.createElement('option');
                opt.value = sch;
                opt.textContent = sch;
                if (sch === currentSelectedSchool) opt.selected = true;
                schoolSelect.appendChild(opt);
            });

            if (yearSelect) {
                const currentSelectedYear = yearSelect.value;
                yearSelect.innerHTML = '<option value="">全部學年度</option>';
                const years = [...new Set(exams.map(e => e.year))].filter(Boolean).sort((a, b) => parseInt(b) - parseInt(a));
                years.forEach(yr => {
                    const opt = document.createElement('option');
                    opt.value = yr;
                    opt.textContent = `${yr} 學年度`;
                    if (yr === currentSelectedYear) opt.selected = true;
                    yearSelect.appendChild(opt);
                });

        }

        function renderPagination(totalItems) {
            if (!paginationContainer || !pageButtonsWrapper) return;

            if (totalItems === 0) {
                paginationContainer.classList.add('hidden');
                return;
            }
            paginationContainer.classList.remove('hidden');

            const totalPages = Math.ceil(totalItems / ITEMS_PER_PAGE);
            if (currentPage > totalPages) currentPage = totalPages || 1;

            const startIdx = (currentPage - 1) * ITEMS_PER_PAGE + 1;
            const endIdx = Math.min(currentPage * ITEMS_PER_PAGE, totalItems);

            if (pageStartIdx) pageStartIdx.textContent = startIdx;
            if (pageEndIdx) pageEndIdx.textContent = endIdx;
            if (pageTotalCount) pageTotalCount.textContent = totalItems;

            pageButtonsWrapper.innerHTML = '';

            const prevBtn = document.createElement('button');
            prevBtn.className = `px-3 py-1.5 rounded-xl text-xs font-semibold transition flex items-center gap-1 shrink-0 ${currentPage === 1 ? 'text-slate-300 cursor-not-allowed' : 'text-slate-600 hover:bg-slate-100 hover:text-indigo-600'}`;
            prevBtn.innerHTML = `<i class="fa-solid fa-chevron-left text-[10px]"></i> 上一頁`;
            prevBtn.disabled = currentPage === 1;
            prevBtn.onclick = () => {
                if (currentPage > 1) {
                    currentPage--;
                    renderExams();
                    window.scrollTo({ top: 0, behavior: 'smooth' });
                }
            };
            pageButtonsWrapper.appendChild(prevBtn);

            const pageNumbers = [];
            if (totalPages <= 7) {
                for (let i = 1; i <= totalPages; i++) pageNumbers.push(i);
            } else {
                if (currentPage <= 4) pageNumbers.push(1, 2, 3, 4, 5, '...', totalPages);
                else if (currentPage >= totalPages - 3) pageNumbers.push(1, '...', totalPages - 4, totalPages - 3, totalPages - 2, totalPages - 1, totalPages);
                else pageNumbers.push(1, '...', currentPage - 1, currentPage, currentPage + 1, '...', totalPages);
            }

            pageNumbers.forEach(p => {
                if (p === '...') {
                    const dotsSpan = document.createElement('span');
                    dotsSpan.className = 'px-1.5 text-xs text-slate-400 font-bold select-none shrink-0';
                    dotsSpan.textContent = '...';
                    pageButtonsWrapper.appendChild(dotsSpan);
                } else {
                    const numBtn = document.createElement('button');
                    const isActive = p === currentPage;
                    numBtn.className = `w-8 h-8 rounded-xl text-xs font-bold transition shrink-0 ${isActive ? 'bg-indigo-600 text-white shadow-md shadow-indigo-500/20' : 'text-slate-600 hover:bg-slate-100 hover:text-indigo-600'}`;
                    numBtn.textContent = p;
                    numBtn.onclick = () => {
                        currentPage = p;
                        renderExams();
                        window.scrollTo({ top: 0, behavior: 'smooth' });
                    };
                    pageButtonsWrapper.appendChild(numBtn);
                }
            });

            const nextBtn = document.createElement('button');
            nextBtn.className = `px-3 py-1.5 rounded-xl text-xs font-semibold transition flex items-center gap-1 shrink-0 ${currentPage === totalPages ? 'text-slate-300 cursor-not-allowed' : 'text-slate-600 hover:bg-slate-100 hover:text-indigo-600'}`;
            nextBtn.innerHTML = `下一頁 <i class="fa-solid fa-chevron-right text-[10px]"></i>`;
            nextBtn.disabled = currentPage === totalPages;
            nextBtn.onclick = () => {
                if (currentPage < totalPages) {
                    currentPage++;
                    renderExams();
                    window.scrollTo({ top: 0, behavior: 'smooth' });
                }
            };
            pageButtonsWrapper.appendChild(nextBtn);
        }

                if (levelVal && item.educationLevel !== levelVal) return false;
                if (cityVal && item.city !== cityVal) return false;
                if (schVal && item.school !== schVal) return false;
                if (grVal && item.grade !== grVal) return false;
                if (semVal && item.semester !== semVal) return false;
                if (subVal && item.subject !== subVal) return false;

                if (yearVal && item.year !== yearVal) return false;

                if (statusVal === 'missing_answer' && hasAnswer) return false;

                    const str = `${item.year} ${item.educationLevel} ${item.city} ${item.school} ${item.grade} ${item.semester} ${item.subject} ${item.category}`.toLowerCase();
                    if (!str.includes(searchVal)) return false;
                }
                return true;

            currentFilteredExams.sort((a, b) => {
                return sortVal === 'newest' ? parseInt(b.year) - parseInt(a.year) : parseInt(a.year) - parseInt(b.year);

        function switchPreviewTab(fileType) {

            modalTitle.textContent = `${currentPreviewExam.school} - ${currentPreviewExam.grade} ${currentPreviewExam.subject} (${typeLabels[fileType]})`;
            modalDownloadBtn.href = fileObj.url;

                embedUrl = fileObj.url.replace(/\/view.*$/, '/preview');

            pdfFrame.src = embedUrl;

            modalPrintBtn.onclick = () => {
                try { pdfFrame.contentWindow.print(); } catch (e) { window.open(fileObj.url, '_blank'); }
            };

            const tabConfigs = [
                { btn: tabBtnQuestion, type: 'question', activeClass: 'bg-white text-indigo-700 shadow-sm' },
                { btn: tabBtnAnswerKey, type: 'answerKey', activeClass: 'bg-white text-emerald-700 shadow-sm' },
                { btn: tabBtnAnswerSheet, type: 'answerSheet', activeClass: 'bg-white text-amber-700 shadow-sm' }
            ];

            tabConfigs.forEach(({ btn, type, activeClass }) => {
                const hasFile = !!currentPreviewExam.files[type];
                btn.disabled = !hasFile;
                btn.className = 'px-3 py-1.5 rounded-lg text-xs font-bold transition flex items-center gap-1.5 ';
                if (!hasFile) btn.className += 'text-slate-400/60 cursor-not-allowed opacity-50';
                else if (type === fileType) btn.className += activeClass;
                else btn.className += 'text-slate-600 hover:text-slate-900 hover:bg-white/50 cursor-pointer';
            });
        }

        tabBtnQuestion?.addEventListener('click', () => switchPreviewTab('question'));
        tabBtnAnswerKey?.addEventListener('click', () => switchPreviewTab('answerKey'));
        tabBtnAnswerSheet?.addEventListener('click', () => switchPreviewTab('answerSheet'));

        function closePreview() {
            previewModal.classList.add('opacity-0', 'pointer-events-none');
            pdfFrame.src = 'about:blank';
            currentPreviewExam = null;
        }

        function handleFilterChange() {
            currentPage = 1;
            renderExams();
        }

        function updateBatchUI() {
            const count = selectedExamKeys.size;
            if (batchCountSpan) batchCountSpan.textContent = count;
            if (count > 0) batchBar.classList.remove('translate-y-24', 'opacity-0');
            else batchBar.classList.add('translate-y-24', 'opacity-0');

            if (selectAllCb) {
                const currentPagesIds = currentFilteredExams.slice((currentPage - 1) * ITEMS_PER_PAGE, currentPage * ITEMS_PER_PAGE).map(item => item.id);
                selectAllCb.checked = currentPagesIds.length > 0 && currentPagesIds.every(id => selectedExamKeys.has(id));
            }
        }

        selectAllCb?.addEventListener('change', (e) => {
            const currentPagesExams = currentFilteredExams.slice((currentPage - 1) * ITEMS_PER_PAGE, currentPage * ITEMS_PER_PAGE);
            currentPagesExams.forEach(item => {
                if (e.target.checked) selectedExamKeys.add(item.id);
                else selectedExamKeys.delete(item.id);
            });
            renderExams();
        });

        function openBatchModal() {
            batchListContainer.innerHTML = '';
            const selectedExams = exams.filter(e => selectedExamKeys.has(e.id));
            let allPdfFiles = [];

            selectedExams.forEach(item => {
                const itemDiv = document.createElement('div');
                itemDiv.className = 'p-4 bg-slate-50 rounded-xl border border-slate-200/80 space-y-2.5';
                let btnsHtml = '';
                if (item.files.question) {
                    allPdfFiles.push({ name: `${item.school} 試題卷`, url: item.files.question.url });
                    btnsHtml += `<a href="${item.files.question.url}" target="_blank" class="px-3 py-1.5 bg-indigo-50 hover:bg-indigo-100 text-indigo-700 rounded-lg text-xs font-semibold transition border border-indigo-200 inline-flex items-center gap-1.5"><i class="fa-regular fa-file-pdf"></i> 試題卷</a>`;
                }
                if (item.files.answerKey) {
                    allPdfFiles.push({ name: `${item.school} 解答`, url: item.files.answerKey.url });
                    btnsHtml += `<a href="${item.files.answerKey.url}" target="_blank" class="px-3 py-1.5 bg-emerald-50 hover:bg-emerald-100 text-emerald-700 rounded-lg text-xs font-semibold transition border border-emerald-200 inline-flex items-center gap-1.5"><i class="fa-solid fa-circle-check"></i> 解答</a>`;
                }
                if (item.files.answerSheet) {
                    allPdfFiles.push({ name: `${item.school} 答案卷`, url: item.files.answerSheet.url });
                    btnsHtml += `<a href="${item.files.answerSheet.url}" target="_blank" class="px-3 py-1.5 bg-amber-50 hover:bg-amber-100 text-amber-700 rounded-lg text-xs font-semibold transition border border-amber-200 inline-flex items-center gap-1.5"><i class="fa-regular fa-copy"></i> 答案卷</a>`;
                }
                itemDiv.innerHTML = `
                    <div class="font-bold text-slate-800 text-sm flex items-center justify-between">
                        <span>${item.school} - ${item.semester} ${item.grade} ${item.subject} ${item.category}</span>
                        <span class="text-xs font-semibold px-2 py-0.5 bg-indigo-50 text-indigo-600 rounded">${item.year}學年</span>
                    </div>
                    <div class="flex flex-wrap gap-2">${btnsHtml}</div>
                `;
                batchListContainer.appendChild(itemDiv);
            });

            copyLinksBtn.onclick = () => {
                const textToCopy = allPdfFiles.map(f => `${f.name}: ${f.url}`).join('\n');
                const tempTextArea = document.createElement('textarea');
                tempTextArea.value = textToCopy;
                document.body.appendChild(tempTextArea);
                tempTextArea.select();
                document.execCommand('copy');
                document.body.removeChild(tempTextArea);
                alert('已複製所有選取的 PDF 連結至剪貼簿！');
            };

            openAllPrintBtn.onclick = () => allPdfFiles.forEach(f => window.open(f.url, '_blank'));
            batchModal.classList.remove('opacity-0', 'pointer-events-none');
        }

        function closeBatchModal() { batchModal.classList.add('opacity-0', 'pointer-events-none'); }

        protectionToggleBtn?.addEventListener('click', () => {
            isProtectionEnabled = !isProtectionEnabled;
            if (isProtectionEnabled) {
                pdfContainer.classList.add('preview-protected');
                protectionStatusText.textContent = '防護: 開';
            } else {
                pdfContainer.classList.remove('preview-protected');
                protectionStatusText.textContent = '防護: 關';
            }
        });

        document.addEventListener('contextmenu', (e) => {
            if (isProtectionEnabled && previewModal && !previewModal.classList.contains('pointer-events-none')) e.preventDefault();
        });

        batchPrintBtn.addEventListener('click', openBatchModal);
        closeBatchModalBtn.addEventListener('click', closeBatchModal);
        closeModalBtn.addEventListener('click', closePreview);
        refreshDataBtn.addEventListener('click', () => loadExamsFromSheet(true));

        batchDownloadBtn.addEventListener('click', () => {
            exams.filter(e => selectedExamKeys.has(e.id)).forEach(item => {
                if (item.files.question) window.open(item.files.question.url, '_blank');
                if (item.files.answerKey) window.open(item.files.answerKey.url, '_blank');
                if (item.files.answerSheet) window.open(item.files.answerSheet.url, '_blank');
            });
        });

        clearBatchBtn.addEventListener('click', () => { selectedExamKeys.clear(); renderExams(); });

        resetBtn.addEventListener('click', () => {
            if (levelSelect) levelSelect.value = '';
            citySelect.value = ''; schoolSelect.value = ''; gradeSelect.value = '';
            semesterSelect.value = ''; subjectSelect.value = ''; categorySelect.value = '';
            if (statusSelect) statusSelect.value = ''; if (yearSelect) yearSelect.value = '';
            isRecentTwoYearsOnly = false;
            if (recentYearsBtn) recentYearsBtn.className = 'text-[11px] font-semibold px-2 py-0.5 rounded-md bg-indigo-50 text-indigo-600 hover:bg-indigo-100 border border-indigo-200/70 transition flex items-center gap-1';
            searchInput.value = ''; sortSelect.value = 'newest';
            selectedExamKeys.clear(); currentPage = 1;
            updateGradeAndSchoolOptions();
            renderExams();
        });

        recentYearsBtn?.addEventListener('click', () => {
            isRecentTwoYearsOnly = !isRecentTwoYearsOnly;
            if (isRecentTwoYearsOnly && yearSelect) yearSelect.value = '';
            if (recentYearsBtn) {
                recentYearsBtn.className = isRecentTwoYearsOnly 
                    ? 'text-[11px] font-bold px-2 py-0.5 rounded-md bg-indigo-600 text-white shadow-sm transition flex items-center gap-1'
                    : 'text-[11px] font-semibold px-2 py-0.5 rounded-md bg-indigo-50 text-indigo-600 hover:bg-indigo-100 border border-indigo-200/70 transition flex items-center gap-1';
            }
            handleFilterChange();
        });

        levelSelect?.addEventListener('change', () => { updateGradeAndSchoolOptions(); handleFilterChange(); });
        citySelect?.addEventListener('change', () => { updateGradeAndSchoolOptions(); handleFilterChange(); });
        yearSelect?.addEventListener('change', () => {
            if (yearSelect.value && isRecentTwoYearsOnly) {
                isRecentTwoYearsOnly = false;
                if (recentYearsBtn) recentYearsBtn.className = 'text-[11px] font-semibold px-2 py-0.5 rounded-md bg-indigo-50 text-indigo-600 hover:bg-indigo-100 border border-indigo-200/70 transition flex items-center gap-1';
            }
            handleFilterChange();
        });

        [schoolSelect, gradeSelect, semesterSelect, subjectSelect, categorySelect, statusSelect, sortSelect].forEach(select => {
            select?.addEventListener('change', handleFilterChange);
        });
        searchInput?.addEventListener('input', handleFilterChange);

        // 頁面載入時直接呼叫 API 取得雲端考卷資料
        window.addEventListener('DOMContentLoaded', () => {
            updateGradeAndSchoolOptions();
            renderExams();
            loadExamsFromSheet(false);
        });
    </script>
