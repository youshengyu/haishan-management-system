// js/api.js
const GAS_API_URL = "https://script.google.com/macros/s/AKfycbywtZ60Z-Goktbygaf182xpY2Srowk_gJFOSB4vlPQn-zoGAGYs8DxupywhbhWNfdH7/exec";

// 共用 GET 請求函式
async function callGasGet(action) {
  try {
    let response = await fetch(`${GAS_API_URL}?action=${action}`);
    let result = await response.json();
    return result;
  } catch (err) {
    console.error("GET 請求失敗:", err);
    return { status: "error", message: err.toString() };
  }
}

// 共用 POST 請求函式 (用於新增、修改、刪除)
async function callGasPost(action, payload) {
  try {
    let response = await fetch(GAS_API_URL, {
      method: "POST",
      body: JSON.stringify({ action: action, payload: payload })
    });
    let result = await response.json();
    return result;
  } catch (err) {
    console.error("POST 請求失敗:", err);
    return { status: "error", message: err.toString() };
  }
}
